function openAction(type){const f={task:["New Estate Task",[["title","Task title","text",""],["department","Department","text","Estate"],["priority","Priority","select",["NORMAL","HIGH","CRITICAL"]],["notes","Notes","textarea",""]],saveTask],diary:["New Estate Diary Entry",[["title","Activity","text",""],["date","Date","date",todayISO()],["time","Time","time",new Date().toTimeString().slice(0,5)],["department","Department","text","Estate"],["owner","Owner","text","Estate Team"]],saveDiary],animal:["Add Animal",[["species","Species","select",["Cattle","Sheep","Poultry"]],["breed","Breed","text",""],["sex","Sex","select",["Female","Male"]],["location","Location","text",""]],saveAnimal],eggs:["Record Heritage Egg Collection",[["collection","Collection","select",["Spring Heritage Egg Collection","Summer Heritage Egg Collection","Harvest Heritage Egg Collection","Christmas Estate Heritage Egg Collection","Royal Estate / Heritage Reserve Collection"]],["whitecream","White / Cream","number","0"],["ivory","Ivory","number","0"],["palebeige","Pale Beige","number","0"],["tan","Tan","number","0"],["richbrown","Rich Brown","number","0"],["darkchocolate","Dark Chocolate","number","0"],["blue","Blue","number","0"],["bluegreen","Blue Green","number","0"],["olivegreen","Olive Green","number","0"],["speckled","Speckled","number","0"]],saveEggs],work:["New Work Order",[["title","Work order","text",""],["asset","Asset / Location","text",""],["priority","Priority","select",["NORMAL","HIGH","CRITICAL"]]],saveWork],stock:["Record Stock Receipt",[["name","Item","text",""],["quantity","Quantity received","number",""]],saveStock]}[type];modal(f[0],`<form id="actionForm" onsubmit="event.preventDefault();window.__saveAction()">${formFields(f[1])}<div class="modal-actions"><button type="button" class="secondary" onclick="closeModal()">Cancel</button><button class="primary" type="submit">Save</button></div></form>`);window.__saveAction=f[2]}
function formFields(fields){return `<div class="form-grid">${fields.map(([id,label,type,val])=>type==="select"?`<div class="field"><label>${label}</label><select id="f_${id}">${val.map(v=>`<option>${v}</option>`).join("")}</select></div>`:type==="textarea"?`<div class="field full"><label>${label}</label><textarea id="f_${id}">${val||""}</textarea></div>`:`<div class="field"><label>${label}</label><input id="f_${id}" type="${type}" value="${esc(val)}"></div>`).join("")}</div>`}
const val=id=>document.getElementById("f_"+id)?.value||"";
function saveTask(){state.tasks.unshift({id:"T"+Date.now(),title:val("title"),department:val("department"),priority:val("priority"),status:"OPEN"});audit("TASK_CREATED",val("title"));closeModal();render()}
function saveDiary(){state.diary.unshift({id:"D"+Date.now(),title:val("title"),date:val("date"),time:val("time"),department:val("department"),owner:val("owner")});audit("DIARY_CREATED",val("title"));closeModal();render()}
function saveAnimal(){state.animals.unshift({id:"CC-A-"+String(state.animals.length+1).padStart(4,"0"),species:val("species"),breed:val("breed"),sex:val("sex"),location:val("location"),status:"ACTIVE"});audit("ANIMAL_CREATED",val("species")+" "+val("breed"));closeModal();render()}
function saveEggs(){const colours={"White / Cream":Number(val("whitecream")||0),"Ivory":Number(val("ivory")||0),"Pale Beige":Number(val("palebeige")||0),"Tan":Number(val("tan")||0),"Rich Brown":Number(val("richbrown")||0),"Dark Chocolate":Number(val("darkchocolate")||0),"Blue":Number(val("blue")||0),"Blue Green":Number(val("bluegreen")||0),"Olive Green":Number(val("olivegreen")||0),"Speckled":Number(val("speckled")||0)};const quantity=Object.values(colours).reduce((a,b)=>a+b,0);state.eggCollections.unshift({id:"E"+Date.now(),collection:val("collection"),quantity,colours,date:todayISO()});audit("EGG_COLLECTION",val("collection")+" · "+quantity+" eggs");closeModal();render();setTimeout(showEggDashboard,20)}
function saveWork(){state.maintenance.unshift({id:"M"+Date.now(),title:val("title"),asset:val("asset"),priority:val("priority"),status:"OPEN"});audit("WORK_ORDER_CREATED",val("title"));closeModal();render()}
function saveStock(){const name=val("name"),qty=Number(val("quantity")||0);let i=state.inventory.find(x=>x.name.toLowerCase()===name.toLowerCase());if(i)i.stock=Number(i.stock||0)+qty;else state.inventory.unshift({id:"I"+Date.now(),name,stock:qty,reorder:0});audit("STOCK_RECEIPT",name+" +"+qty);closeModal();render()}
function openDiaryDetail(id){const d=state.diary.find(x=>x.id===id);if(d)modal("Estate Diary Entry",`<p><strong>${esc(d.title)}</strong></p><p>${esc(d.date)} · ${esc(d.time)}</p><p>${esc(d.department)} · ${esc(d.owner)}</p>`)}
function modal(title,body){document.body.insertAdjacentHTML("beforeend",`<div class="modal-backdrop" id="modalBackdrop" onclick="if(event.target===this)closeModal()"><section class="modal"><div class="modal-head"><div><h3>${title}</h3><p>Crown & Cross Estate Management System</p></div><button class="close" onclick="closeModal()">×</button></div>${body}</section></div>`)}
function closeModal(){document.getElementById("modalBackdrop")?.remove()}
render();
